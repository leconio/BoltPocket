cmake_minimum_required(VERSION 3.22)
project(BoltRuntime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(llama)
add_library(boltcore STATIC engine.cpp gpu_probe.cpp llama/vendor/hash/sha256/sha256.c)
# Bind saved state to all native sources and backend build options, not to UI-only APK updates.
file(GLOB_RECURSE BOLT_CACHE_SOURCES CONFIGURE_DEPENDS
    "*.cpp" "*.c" "*.h" "*.hpp" "*.cl" "*.comp" "*.metal" "CMakeLists.txt")
list(SORT BOLT_CACHE_SOURCES)
set(BOLT_CACHE_FINGERPRINT "${GGML_OPENCL}:${GGML_VULKAN}:${GGML_METAL}:${CMAKE_CXX_COMPILER_ID}:${CMAKE_CXX_COMPILER_VERSION}:${CMAKE_SYSTEM_PROCESSOR}")
foreach(BOLT_CACHE_SOURCE IN LISTS BOLT_CACHE_SOURCES)
    file(SHA256 "${BOLT_CACHE_SOURCE}" BOLT_CACHE_SOURCE_HASH)
    file(RELATIVE_PATH BOLT_CACHE_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${BOLT_CACHE_SOURCE}")
    string(APPEND BOLT_CACHE_FINGERPRINT "${BOLT_CACHE_RELATIVE}:${BOLT_CACHE_SOURCE_HASH}\n")
endforeach()
string(SHA256 BOLT_CACHE_BUILD_ID "${BOLT_CACHE_FINGERPRINT}")
target_compile_definitions(boltcore PRIVATE BOLT_CACHE_BUILD_ID="${BOLT_CACHE_BUILD_ID}")
if(GGML_VULKAN)
  target_compile_definitions(boltcore PRIVATE BOLT_VULKAN=1)
endif()
if(GGML_OPENCL)
  target_compile_definitions(boltcore PRIVATE BOLT_OPENCL=1)
endif()
target_include_directories(boltcore PUBLIC . llama/common PRIVATE llama/vendor/hash)
target_link_libraries(boltcore PUBLIC llama llama-common)
if(ANDROID)
  add_library(boltexpo SHARED jni.cpp)
  target_link_libraries(boltexpo PRIVATE boltcore log android)
endif()
